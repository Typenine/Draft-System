import { NextRequest, NextResponse } from 'next/server';
import { verifyRequestSession } from '@/lib/auth-server';
import { databaseConfigured, DatabaseNotConfiguredError } from '@/lib/db';
import { getDraftLiveState, getDraftState } from '@/lib/store';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

const ACTIVE_PAUSE_REASONS = new Set(['pending_pick', 'pick_animation', 'trade_animation', 'end_draft_animation', 'round_end']);

function stateCacheControl(session: unknown, state: Awaited<ReturnType<typeof getDraftLiveState>>): string {
  if (session) return 'private, no-store';
  const active = state.draft?.status === 'LIVE'
    || (state.draft?.status === 'PAUSED' && ACTIVE_PAUSE_REASONS.has(state.draft.pauseReason || ''));
  return active
    ? 'public, max-age=0, s-maxage=2, stale-while-revalidate=5'
    : 'public, max-age=0, s-maxage=900, stale-while-revalidate=60';
}

export async function GET(req: NextRequest) {
  if (!databaseConfigured()) {
    return NextResponse.json({
      configured: false,
      databaseConfigured: false,
      error: 'DATABASE_URL is not configured.',
      draft: null,
      teams: [],
      players: [],
      slots: [],
      picks: [],
      currentTeam: null,
      availablePlayers: [],
    }, { status: 503 });
  }
  try {
    const liveMode = req.nextUrl.searchParams.get('mode') === 'live';
    const session = await verifyRequestSession(req);
    // Anonymous callers only get the lightweight state. The expensive player pool
    // and available-player calculation are reserved for authenticated draft users.
    const state = liveMode || !session
      ? await getDraftLiveState({ includeModeration: session?.role === 'admin' })
      : await getDraftState();
    const ownPendingPick = session?.role === 'team' && state.pendingPick?.teamId === session.teamId ? state.pendingPick : null;
    const pendingPick = session?.role === 'admin' ? state.pendingPick : ownPendingPick;
    const pendingTrades = session?.role === 'admin' ? state.pendingTrades : [];

    if (liveMode || !session) {
      const { players: _players, availablePlayers: _availablePlayers, ...liveState } = state;
      return NextResponse.json({ ...liveState, pendingPick, pendingTrades }, { headers: { 'Cache-Control': stateCacheControl(session, state) } });
    }
    if (session.role === 'admin') return NextResponse.json(state, { headers: { 'Cache-Control': 'private, no-store' } });
    return NextResponse.json({
      ...state,
      players: state.players,
      availablePlayers: state.availablePlayers,
      pendingPick,
      pendingTrades,
    }, { headers: { 'Cache-Control': 'private, no-store' } });
  } catch (error) {
    const message = error instanceof Error ? error.message : 'state_load_failed';
    return NextResponse.json({ error: message }, { status: error instanceof DatabaseNotConfiguredError ? 503 : 500 });
  }
}
