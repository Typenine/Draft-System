import { NextRequest, NextResponse } from 'next/server';
import { verifyRequestSession } from '@/lib/auth-server';
import { ensureSchema, getSql } from '@/lib/db';
import { rowsOf, type Row } from '@/lib/store/shared';

export const runtime = 'nodejs';
export const dynamic = 'force-dynamic';

function json(data: unknown, status = 200) {
  return NextResponse.json(data, { status, headers: { 'Cache-Control': 'no-store' } });
}
function validMediaUrl(value: unknown): string | null {
  const url = typeof value === 'string' ? value.trim() : '';
  if (!url) return null;
  if (url.startsWith('/') || url.startsWith('https://')) return url;
  throw new Error('media_url_must_be_https_or_public_path');
}

export async function GET() {
  try {
    await ensureSchema();
    const rows = rowsOf<Row>(await getSql()`
      SELECT player_id, player_name, image_url, video_url
      FROM draft_player_media
      WHERE image_url IS NOT NULL OR video_url IS NOT NULL
      ORDER BY COALESCE(player_name, player_id)
    `);
    return json({ videos: rows.map((row) => ({
      playerId: String(row.player_id),
      playerName: row.player_name ? String(row.player_name) : null,
      videoUrl: row.video_url ? String(row.video_url) : null,
      imageUrl: row.image_url ? String(row.image_url) : null,
      hasImage: Boolean(row.image_url),
    })) });
  } catch {
    return json({ videos: [] });
  }
}

export async function POST(req: NextRequest) {
  const session = await verifyRequestSession(req);
  if (!session || session.role !== 'admin') return json({ error: 'commissioner_required' }, 403);
  try {
    await ensureSchema();
    const body = await req.json() as Record<string, unknown>;
    const playerId = String(body.playerId || '').trim();
    if (!playerId) return json({ error: 'player_id_required' }, 400);
    const player = rowsOf<Row>(await getSql()`SELECT id, name FROM draft_players WHERE id = ${playerId} LIMIT 1`)[0];
    if (!player) return json({ error: 'player_not_found' }, 404);
    if (body.action === 'delete') {
      const mediaType = String(body.type || 'video');
      if (mediaType === 'image') await getSql()`UPDATE draft_player_media SET image_url = NULL, updated_at = now() WHERE player_id = ${playerId}`;
      else if (mediaType === 'all') await getSql()`DELETE FROM draft_player_media WHERE player_id = ${playerId}`;
      else await getSql()`UPDATE draft_player_media SET video_url = NULL, updated_at = now() WHERE player_id = ${playerId}`;
      return json({ ok: true });
    }
    const imageUrl = body.imageUrl === undefined ? undefined : validMediaUrl(body.imageUrl);
    const videoUrl = body.videoUrl === undefined ? undefined : validMediaUrl(body.videoUrl);
    if (imageUrl === undefined && videoUrl === undefined) return json({ error: 'image_or_video_url_required' }, 400);
    await getSql()`
      INSERT INTO draft_player_media (player_id, player_name, image_url, video_url, updated_at)
      VALUES (${playerId}, ${String(player.name)}, ${imageUrl ?? null}, ${videoUrl ?? null}, now())
      ON CONFLICT (player_id) DO UPDATE SET
        player_name = EXCLUDED.player_name,
        image_url = CASE WHEN ${imageUrl === undefined} THEN draft_player_media.image_url ELSE EXCLUDED.image_url END,
        video_url = CASE WHEN ${videoUrl === undefined} THEN draft_player_media.video_url ELSE EXCLUDED.video_url END,
        updated_at = now()
    `;
    return json({ ok: true });
  } catch (error) {
    return json({ error: error instanceof Error ? error.message : 'media_save_failed' }, 400);
  }
}
