'use client';

import { useEffect, useRef } from 'react';
import { gsap } from 'gsap';

type EndOfDraftAnimationProps = {
  eventName?: string | null;
  eventYear?: number | null;
  eventLogoUrl?: string | null;
  eventColor1?: string | null;
  totalPicks: number;
  totalRounds: number;
  onComplete: () => void;
};

export default function EndOfDraftAnimation({
  eventName,
  eventYear,
  eventLogoUrl,
  eventColor1,
  totalPicks,
  totalRounds,
  onComplete,
}: EndOfDraftAnimationProps) {
  const rootRef = useRef<HTMLDivElement>(null);
  const logoRef = useRef<HTMLImageElement>(null);
  const eyebrowRef = useRef<HTMLDivElement>(null);
  const titleRef = useRef<HTMLHeadingElement>(null);
  const detailRef = useRef<HTMLDivElement>(null);
  const completeRef = useRef(onComplete);
  completeRef.current = onComplete;
  const accent = eventColor1 || '#facc15';

  useEffect(() => {
    const root = rootRef.current;
    if (!root) return;
    const timeline = gsap.timeline({ onComplete: () => completeRef.current() });
    timeline
      .fromTo(root, { opacity: 0 }, { opacity: 1, duration: 0.55, ease: 'power2.out' })
      .fromTo(logoRef.current, { opacity: 0, scale: 0.72, rotate: -5 }, { opacity: 1, scale: 1, rotate: 0, duration: 0.9, ease: 'back.out(1.5)' }, 0.15)
      .fromTo(eyebrowRef.current, { opacity: 0, y: 20 }, { opacity: 1, y: 0, duration: 0.45 }, 0.55)
      .fromTo(titleRef.current, { opacity: 0, y: 34, scale: 0.94 }, { opacity: 1, y: 0, scale: 1, duration: 0.75, ease: 'power3.out' }, 0.72)
      .fromTo(detailRef.current, { opacity: 0, y: 18 }, { opacity: 1, y: 0, duration: 0.5 }, 1.15)
      .to({}, { duration: 5.4 })
      .to([detailRef.current, titleRef.current, eyebrowRef.current, logoRef.current], { opacity: 0, y: -18, duration: 0.55, stagger: 0.05, ease: 'power2.in' })
      .to(root, { opacity: 0, duration: 0.45 });
    return () => { timeline.kill(); };
  }, []);

  return (
    <div ref={rootRef} className="absolute inset-0 z-[120] overflow-hidden bg-zinc-950 flex items-center justify-center" style={{ opacity: 0 }}>
      <div className="absolute inset-0" style={{ background: `radial-gradient(circle at 50% 36%, ${accent}33 0%, transparent 38%), linear-gradient(135deg, #020617 0%, #09090b 55%, #111827 100%)` }} />
      <div className="absolute inset-x-0 top-0 h-2" style={{ background: accent, boxShadow: `0 0 36px ${accent}` }} />
      <div className="absolute inset-x-0 bottom-0 h-2" style={{ background: accent, boxShadow: `0 0 36px ${accent}` }} />
      <div className="relative z-10 flex max-w-5xl flex-col items-center px-8 text-center">
        {eventLogoUrl && <img ref={logoRef} src={eventLogoUrl} alt="" className="mb-5 h-52 w-52 object-contain drop-shadow-2xl" />}
        <div ref={eyebrowRef} className="text-sm font-black uppercase tracking-[0.42em] text-zinc-300">
          {eventName || 'Draft'}{eventYear ? ` · ${eventYear}` : ''}
        </div>
        <h2 ref={titleRef} className="mt-3 text-7xl font-black uppercase tracking-tight text-white md:text-8xl" style={{ textShadow: `0 0 34px ${accent}66` }}>
          Draft Complete
        </h2>
        <div ref={detailRef} className="mt-5 rounded-full border border-white/20 bg-black/35 px-8 py-3 text-xl font-bold text-zinc-100 backdrop-blur-sm">
          {totalPicks} selections finalized · {totalRounds} rounds complete
        </div>
      </div>
    </div>
  );
}
